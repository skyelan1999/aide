func TestMemoryNeuralTokenPersistence(t *testing.T) {
	a := testApp(t)
	for _, text := range []string{"我喜欢中文", "不要删除文档", "中文参数版本"} {
		if got := a.writeMemory(text); !strings.Contains(got, "已写入 token") {
			t.Fatal(got)
		}
	}
	s, warning, err := loadMemoryTokenStore(a.projectCacheDir())
	if err != nil || warning != "" || len(s.Records) != 3 {
		t.Fatalf("%+v %s %v", s, warning, err)
	}
	if !strings.Contains(a.readMemory(), "不要删除文档") {
		t.Fatal(a.readMemory())
	}
	g := memoryNeuralGraph(s)
	if g["tokenizerName"] != "Unicode 字符级" || g["tokenizerDescription"] != "本地 Unicode 标量分词；非模型 BPE" {
		t.Fatalf("missing tokenizer metadata: %+v", g)
	}
	found := false
	for _, n := range g["nodes"].([]*memoryNeuron) {
		if n.Token == "文" {
			found = true
			if n.Records != 3 || n.Regions["preferences"] == 0 || n.Regions["constraints"] == 0 || n.Regions["facts"] == 0 {
				t.Fatalf("%+v", n)
			}
		}
	}
	if !found {
		t.Fatal("missing shared neuron")
	}
	requireStatus(t, request(a, "GET", "/api/memory-neural", nil), 200)
	// Side-by-side compatibility: VoiceAgent can still read text, never tokens
	// or its own private memory through this endpoint.
	os.MkdirAll(filepath.Join(a.dataPath, "assistant"), 0700)
	os.WriteFile(filepath.Join(a.dataPath, "assistant", "voice-memory.json"), []byte("PRIVATE_NEVER_SHOW"), 0600)
	w := request(a, "GET", "/api/memory-neural", nil)
	if strings.Contains(w.Body.String(), "PRIVATE_NEVER_SHOW") {
		t.Fatal("private memory exposed")
	}
}
func TestMemoryNeuralLegacyAndExternalEdit(t *testing.T) {
	a := testApp(t)
	dir := a.projectCacheDir()
	os.MkdirAll(dir, 0700)
	os.WriteFile(a.memoryPath(), []byte("旧记录🙂\n"), 0600)
	if got := a.writeMemory("新增记录"); !strings.Contains(got, "已写入") {
		t.Fatal(got)
	}
	s, _, err := loadMemoryTokenStore(dir)
	if err != nil || len(s.Records) != 2 || s.Records[0].Origin != "legacy-text" {
		t.Fatalf("%+v %v", s, err)
	}
	os.WriteFile(a.memoryPath(), []byte("替换记忆"), 0600)
	s, warning, err := loadMemoryTokenStore(dir)
	if err != nil || warning == "" || s.text() != "替换记忆" {
		t.Fatalf("%+v %s %v", s, warning, err)
	}
	if got := a.writeMemory("后续"); !strings.Contains(got, "已写入") {
		t.Fatal(got)
	}
	if strings.Contains(a.readMemory(), "旧记录") {
		t.Fatal("deleted memory resurrected")
	}
	// Missing compatibility projection keeps JSON as source.
	os.Remove(a.memoryPath())
	s, _, err = loadMemoryTokenStore(dir)
	if err != nil || !strings.Contains(s.text(), "后续") {
		t.Fatal(err)
	}
}
func TestMemoryNeuralCorruptionAndLimits(t *testing.T) {
	a := testApp(t)
	a.writeMemory("正常")
	before, _ := os.ReadFile(a.memoryPath())
	if got := a.writeMemory(strings.Repeat("a", memoryTextLimit)); !strings.Contains(got, "超过") {
		t.Fatal(got)
	}
	after, _ := os.ReadFile(a.memoryPath())
	if string(after) != string(before) {
		t.Fatal("over limit mutated")
	}
	os.WriteFile(filepath.Join(a.projectCacheDir(), memoryTokenFile), []byte("bad"), 0600)
	if got := a.writeMemory("不得覆盖"); !strings.Contains(got, "失败") {
		t.Fatal(got)
	}
	requireStatus(t, request(a, "GET", "/api/memory-neural", nil), 500)
	s := newMemoryTokenStore()
	s.Tokenizer = "unregistered"
	if s.validate() == nil {
		t.Fatal("unsupported tokenizer")
	}
	s = newMemoryTokenStore()
	s.append("abcd", "test")
	delete(s.Vocabulary, s.Records[0].Tokens[0])
	if s.validate() == nil {
		t.Fatal("missing token accepted")
	}
}
func TestMemoryNeuralLockAndWorkspaceIsolation(t *testing.T) {
	a := testApp(t)
	a.writeMemory("workspace-A")
	a.mu.Lock()
	a.settings.UserPasswordHash = "configured"
	a.mu.Unlock()
	_, err := a.writeLockState(true)
	if err != nil {
		t.Fatal(err)
	}
	requireStatus(t, request(a, "GET", "/api/memory-neural", nil), http.StatusLocked)
	a.mu.Lock()
	a.settings.UserPasswordHash = ""
	a.mu.Unlock()
	a.writeLockState(false)
	a.mu.Lock()
	a.cacheContainer = filepath.Join(t.TempDir(), "cache")
	a.mu.Unlock()
	w := request(a, "GET", "/api/memory-neural", nil)
	requireStatus(t, w, 200)
	if strings.Contains(w.Body.String(), "workspace-A") {
		t.Fatal("cross-workspace leak")
	}
	var g map[string]any
	if err = json.Unmarshal(w.Body.Bytes(), &g); err != nil {
		t.Fatal(err)
	}
	if g["totalNodes"].(float64) != 0 {
		t.Fatal(g)
	}
}
func TestMemoryNeuralStableLegacyAndUnicode(t *testing.T) {
	dir := t.TempDir()
	os.WriteFile(filepath.Join(dir, "memory.md"), []byte("中文🙂\n不要删除\n"), 0600)
	first, _, err := loadMemoryTokenStore(dir)
	if err != nil {
		t.Fatal(err)
	}
	second, _, err := loadMemoryTokenStore(dir)
	if err != nil {
		t.Fatal(err)
	}
	if first.Records[0].ID != second.Records[0].ID || first.Records[0].Created != second.Records[0].Created {
		t.Fatal("unstable legacy identity")
	}
	if len(first.Records[0].Tokens) != 4 || first.text() != "中文🙂\n不要删除\n" {
		t.Fatal(first)
	}
	g := memoryNeuralGraph(first)
	for _, edge := range g["edges"].([]*memorySynapse) {
		if _, ok := first.Vocabulary[edge.From]; !ok {
			t.Fatal("phantom edge")
		}
	}
}
func TestMemoryNeuralGraphBound(t *testing.T) {
	s := newMemoryTokenStore()
	var b strings.Builder
	for r := rune(0x4000); r < 0x4000+1300; r++ {
		b.WriteRune(r)
	}
	s.append(b.String(), "test")
	g := memoryNeuralGraph(s)
	if g["totalNodes"] != 1300 || len(g["nodes"].([]*memoryNeuron)) != 1200 || g["truncated"] != true {
		t.Fatal("graph cap/coverage")
	}
}
func TestMemoryNeuralRejectsLinkedMemory(t *testing.T) {
	a := testApp(t)
	os.MkdirAll(a.projectCacheDir(), 0700)
	target := filepath.Join(t.TempDir(), "private")
	os.WriteFile(target, []byte("private"), 0600)
	if err := os.Symlink(target, a.memoryPath()); err != nil {
		t.Skip(err)
	}
	requireStatus(t, request(a, "GET", "/api/memory-neural", nil), 500)
	if got := a.writeMemory("safe"); !strings.Contains(got, "失败") {
		t.Fatal(got)
	}
}
