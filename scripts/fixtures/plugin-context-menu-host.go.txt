package server

import (
 "archive/zip"
 "io"
 "net/http/httptest"
 "os"
 "os/exec"
 "path/filepath"
 "testing"
)

// Optional external plugin stays outside the public source tree and image.
func TestPluginContextMenuExternal(t *testing.T) {
 archivePath:=os.Getenv("COMPARE_TEST_ARCHIVE");if archivePath=="" {t.Skip("external plugin UI fixture")}
 a:=testApp(t)
 archive,err:=zip.OpenReader(archivePath);if err!=nil {t.Fatal(err)};defer archive.Close()
 entries:=map[string]string{}
 for _,file:=range archive.File {reader,err:=file.Open();if err!=nil {t.Fatal(err)};data,err:=io.ReadAll(reader);reader.Close();if err!=nil {t.Fatal(err)};entries[file.Name]=string(data)}
 requireStatus(t,pluginBundleRequest(t,a,entries),201)
 for name,content:=range map[string]string{"menu-left.txt":"before\n","menu-right.txt":"after\n"} {if err:=os.WriteFile(filepath.Join(a.workPath,name),[]byte(content),0600);err!=nil {t.Fatal(err)}}
 if err:=os.MkdirAll(filepath.Join(a.dataPath,"auth"),0700);err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(filepath.Join(a.dataPath,"auth/access-token"),[]byte(a.token),0600);err!=nil {t.Fatal(err)}
 srv:=httptest.NewServer(a.Handler());defer srv.Close()
 cmd:=exec.Command("node",filepath.Join(os.Getenv("AIDE_SOURCE"),"scripts/browser-plugin-context-menu.cjs"));cmd.Env=append(os.Environ(),"AIDE_TEST_URL="+srv.URL,"AIDE_TEST_DATA="+a.dataPath)
 output,err:=cmd.CombinedOutput();if err!=nil {t.Fatalf("browser: %v\n%s",err,output)};t.Log(string(output))
}
